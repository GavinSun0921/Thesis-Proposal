% Shared editable vector styles for the proposed Section 3.2 technical routes.
%
% Design rationale (2026-10-28 revision):
% The earlier version drew every stage as a closed rectangle, which made the
% route figures read as tables of boxes rather than as a flow. This version
% drops the full border and gives each stage a numbered step marker, a header
% rule and generous white space, so the eye follows the number sequence. Stages
% that carry the study's own mechanism get a tinted band and a filled marker;
% stages that merely receive or pass information get an open marker. Emphasis is
% therefore carried by fill and marker weight, both of which survive grayscale.
%
% All text stays black; the accent colours are used only for local sketches.

% One stage of the flow: x, width, step number, title and body text.
% The optional first argument marks a stage that carries the study's mechanism.
\newcommand{\trStage}[6][]{%
  \pgfmathsetmacro{\trCx}{#2+#3/2}%
  \pgfmathsetlengthmacro{\trTextW}{(#3-0.34)*1cm}%
  \if\relax\detokenize{#1}\relax
    \draw[black,line width=0.5pt] (#2,\trStageTop) -- ({#2+#3},\trStageTop);
    \node[trMarker] at (\trCx,\trNumY) {#4};
  \else
    \path[fill=trFill] (#2,{\trStageBot-0.10}) rectangle ({#2+#3},{\trStageTop+0.16});
    \draw[black,line width=0.9pt] (#2,\trStageTop) -- ({#2+#3},\trStageTop);
    \node[trMarkerCore] at (\trCx,\trNumY) {#4};
  \fi
  \node[trTitle,text width=\trTextW] at (\trCx,\trTitleY) {#5};
  \node[trNote,text width=\trTextW] at (\trCx,\trBodyY) {#6};
}

% Arrow between two consecutive stages, drawn on the shared flow line.
\newcommand{\trStep}[2]{%
  \draw[trArrow] (#1,\trArrowY) -- (#2,\trArrowY);
}
